function getFilteredReports(){
  let fEmp=document.getElementById('filterEmployee').value;
  let fWeek=document.getElementById('filterWeek').value;
  let fMonth=document.getElementById('filterMonth').value;
  let fFrom=document.getElementById('filterDateFrom').value;
  let fTo=document.getElementById('filterDateTo').value;
  let fStatus=document.getElementById('filterStatus').value;
  let fDaily=document.getElementById('filterDaily').value;

  return reports.filter(r=>{
    if(fEmp!=='ALL' && r.employee!==fEmp) return false;
    if(fWeek!=='ALL' && week(r.date)!==fWeek) return false;
    if(fMonth!=='ALL' && month(r.date)!==fMonth) return false;
    if(fFrom && parseDate(r.date)<new Date(fFrom)) return false;
    if(fTo && parseDate(r.date)>new Date(fTo)) return false;
    if(fStatus!=='ALL' && r.status!==fStatus) return false;
    if(fDaily==='DAILY' && !r.isDaily) return false;
    if(fDaily==='NODAILY' && r.isDaily) return false;
    return true;
  });
}

function renderGoals(){
  let html='';

function populateWorkloadFilters(){
  let weeks=[...new Set(reports.map(r=>week(r.date)))].sort().reverse();
  let wOpts='<option value="ALL">Tất cả</option>';
  weeks.forEach(w=>{wOpts+=`<option value="${w}">${w}</option>`;});
  let elW=document.getElementById('workloadWeek');
  if(elW){
    let sel=elW.value;
    elW.innerHTML=wOpts;
    elW.value=weeks.includes(sel)?sel:'ALL';
  }

  let months=[...new Set(reports.map(r=>month(r.date)))].sort().reverse();
  let mOpts='<option value="ALL">Tất cả</option>';
  months.forEach(m=>{mOpts+=`<option value="${m}">${m}</option>`;});
  let elM=document.getElementById('workloadMonth');
  if(elM){
    let selM=elM.value;
    elM.innerHTML=mOpts;
    elM.value=months.includes(selM)?selM:'ALL';
  }
}

function clearWorkloadFilters(){
  let w=document.getElementById('workloadWeek'); if(w) w.value='ALL';
  let m=document.getElementById('workloadMonth'); if(m) m.value='ALL';
  let f=document.getElementById('workloadFrom'); if(f) f.value='';
  let t=document.getElementById('workloadTo'); if(t) t.value='';
  renderWorkload();
}

function getWorkloadFilteredReports(){
  let wWeek=document.getElementById('workloadWeek')?.value || 'ALL';
  let wMonth=document.getElementById('workloadMonth')?.value || 'ALL';
  let wFrom=document.getElementById('workloadFrom')?.value || '';
  let wTo=document.getElementById('workloadTo')?.value || '';

  return reports.filter(r=>{
    if(wWeek!=='ALL' && week(r.date)!==wWeek) return false;
    if(wMonth!=='ALL' && month(r.date)!==wMonth) return false;
    if(wFrom && parseDate(r.date)<new Date(wFrom)) return false;
    if(wTo && parseDate(r.date)>new Date(wTo)) return false;
    return true;
  });
}

let activeWorkloadDiffFilter='ALL';
function filterWorkload(diff){
  activeWorkloadDiffFilter=activeWorkloadDiffFilter===diff?'ALL':diff;
  renderWorkload();
}

function renderWorkload(){
  let data=getWorkloadFilteredReports();
  let counts={A:0, B:0, C:0, D:0};
  data.forEach(r=>{
    let d=keyDiff(r.difficulty);
    if(counts[d]!==undefined) counts[d]++;
  });

  document.getElementById('countA').textContent=counts.A;
  document.getElementById('countB').textContent=counts.B;
  document.getElementById('countC').textContent=counts.C;
  document.getElementById('countD').textContent=counts.D;

  // Highlight active stat button
  document.querySelectorAll('.stat-btn').forEach(btn=>{
    btn.classList.remove('active');
    if(btn.classList.contains(activeWorkloadDiffFilter)){
      btn.classList.add('active');
    }
  });

  // Group by employee for chart
  let empMap={};
  employees.forEach(e=>{ empMap[e]={A:0, B:0, C:0, D:0, total:0}; });
  data.forEach(r=>{
    if(!empMap[r.employee]){
      empMap[r.employee]={A:0, B:0, C:0, D:0, total:0};
    }
    let d=keyDiff(r.difficulty);
    if(empMap[r.employee][d]!==undefined){
      empMap[r.employee][d]++;
      empMap[r.employee].total++;
    }
  });

  let chartHtml='';
  let maxTotal=Math.max(...Object.values(empMap).map(o=>o.total), 1);

  Object.keys(empMap).forEach(emp=>{
    let obj=empMap[emp];
    if(activeWorkloadDiffFilter!=='ALL' && obj[activeWorkloadDiffFilter]===0) return;

    let pA=(obj.A/maxTotal)*100;
    let pB=(obj.B/maxTotal)*100;
    let pC=(obj.C/maxTotal)*100;
    let pD=(obj.D/maxTotal)*100;

    chartHtml+=`<div class="chart-row" onclick="openWorkloadDetail('${esc(emp)}')">
      <div style="font-size:12px;font-weight:600;white-space:nowrap;overflow:hidden;text-overflow:ellipsis" title="${esc(emp)}">${esc(emp)}</div>
      <div class="bar">
        <div class="segA" style="width:${pA}%" title="A: ${obj.A}"></div>
        <div class="segB" style="width:${pB}%" title="B: ${obj.B}"></div>
        <div class="segC" style="width:${pC}%" title="C: ${obj.C}"></div>
        <div class="segD" style="width:${pD}%" title="D: ${obj.D}"></div>
      </div>
      <div class="chart-num">${obj.total} việc</div>
    </div>`;
  });

  document.getElementById('workloadChart').innerHTML=chartHtml||`<div class="empty">Không có dữ liệu thống kê phù hợp</div>`;
  document.getElementById('workloadDetail').innerHTML='';
}

function openWorkloadDetail(empName){
  let data=getWorkloadFilteredReports().filter(r=>{
    if(r.employee!==empName) return false;
    if(activeWorkloadDiffFilter!=='ALL' && keyDiff(r.difficulty)!==activeWorkloadDiffFilter) return false;
    return true;
  });

  document.getElementById('workloadDetailTitle').textContent=`Chi tiết công việc của: ${empName} ${activeWorkloadDiffFilter!=='ALL'?'(Độ khó: '+activeWorkloadDiffFilter+')':''}`;
  let html='';
  data.forEach((r, idx)=>{
    let dKey=keyDiff(r.difficulty);
    let statusClass=r.status==='Hoàn thành'?'status-done':'status-working';
    html+=`<tr>
      <td>${idx+1}</td>
      <td class="emp-name">${esc(r.employee)}</td>
      <td>${esc(r.date)}</td>
      <td class="task-cell">${esc(r.task)}</td>
      <td><span class="diff ${dKey}">${esc(r.difficulty)}</span></td>
      <td><span class="status-select ${statusClass}">${esc(r.status)}</span></td>
      <td class="comment-cell">${esc(r.employeeComment)}</td>
    </tr>`;
  });
  document.getElementById('workloadDetailBody').innerHTML=html||`<tr><td colspan="7" class="empty">Không có dữ liệu chi tiết</td></tr>`;
  openModal('workloadDetailModal');
}
